# Библиотечная система
Консольное приложение для управления библиотекой. Реализованы операции 
с книгами, читателями, выдача и возврат книг, статистика.


# Установка и настройка
Создание базы данных
В PostgreSQL под пользователем с правами администратора и выполните:

sql
CREATE USER library_user WITH PASSWORD 'secure_password';
CREATE DATABASE library_db OWNER library_user;
GRANT ALL PRIVILEGES ON DATABASE library_db TO library_user;

# Структура проекта
text
src/main/java/com/library
├── exception          #  исключение
├── util               # DatabaseManager, PropertiesLoader
├── domain              # сущности Book, Reader, Loan (Lombok)
├── dao                # интерфейсы и JDBC-реализации
├── service            # бизнес-логика (LibraryService)
├── console            # консольный интерфейс (ConsoleRun)
└── Main               # точка входа
src/main/resources
├── application.properties   # параметры подключения к БД
├── schema.sql               # создание таблиц и индексов
└── test-data.sql            # тестовые данные
